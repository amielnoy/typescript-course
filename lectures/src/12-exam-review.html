      class ApiClient {
        private requests = 0;
        constructor(public readonly baseUrl: string,
                    private token: string) {}

        get(path: string): string {
          this.requests++;
          return `GET ${this.baseUrl}${path}` +
            ` (#${this.requests}, token ${this.token.length} chars)`;
        }
      }
      const api = new ApiClient("https://qa.dev", "secret");
      console.log(api.get("/users"));
      console.log(api.get("/orders"));

      // api.baseUrl = "x";  ❌ read-only
      // api.token;          ❌ private

Abstract class + Template Method

  • abstract class – אי אפשר ליצור ממנה אובייקט עם new
  • abstract body() – כל מחלקה יורשת חייבת לממש
  • מחלקה אבסטרקטית יכולה להכיל גם מתודות רגילות, כמו run() כאן
  • Template Method: ההורה קובע את הסדר (setup → body → teardown), והילד ממלא רק את החלק שמשתנה
        setup → fill form + submit → teardown
      
      abstract class UiTest {
        run(): string {
          const steps = ["setup", this.body(), "teardown"];
          return steps.join(" → ");
        }
        protected abstract body(): string;
      }

      class LoginTest extends UiTest {
        protected body(): string { return "fill form + submit"; }
      }
      console.log(new LoginTest().run());

      // new UiTest();
      // ❌ Cannot create an instance of an abstract class.

      type Selector = string | string[] | { role: string; name: string };

      function toLocator(s: Selector): string {
        if (typeof s === "string") return `css=${s}`;
        if (Array.isArray(s)) return s.map((x) => "css=" + x).join(" >> ");
        // only the object is left
        return `role=${s.role}[name="${s.name}"]`;
      }
      console.log(toLocator("#pay"));
      console.log(toLocator(["form", "button"]));
      console.log(toLocator({ role: "button", name: "Pay" }));

      type Retry = { retryAfterMs: number };
      type GiveUp = { reason: string };
      function nextStep(r: Retry | GiveUp): string {
        return "retryAfterMs" in r
          ? `retry in ${r.retryAfterMs}ms` : `stop: ${r.reason}`;
      }
      console.log(nextStep({ retryAfterMs: 250 }));
      console.log(nextStep({ reason: "quota exceeded" }));

      class HttpError extends Error {
        constructor(public readonly status: number, message: string) {
          super(`HTTP ${status}: ${message}`);
          this.name = "HttpError";
        }
      }

      function describeError(e: unknown): string {
        if (e instanceof HttpError)
          return `${e.name} ${e.status} – ${e.message}`;
        if (e instanceof Error) return e.message;
        return String(e);
      }
      console.log(describeError(new HttpError(503, "unavailable")));
      console.log(describeError(new TypeError("x is undefined")));
      console.log(describeError("socket hang up"));
